/* ==========================================================================
   Body Scent - Shared JavaScript (product.html, order.html, admin.html)
   ========================================================================== */

// แทนที่ URL ด้านล่างด้วย URL จริงของ Google Apps Script และ Google Sheets CSV
const APPS_SCRIPT_URL = '[APPS_SCRIPT_URL]';
const CSV_URL = '[CSV_URL]';

document.addEventListener('DOMContentLoaded', () => {
  // ตรวจสอบแต่ละหน้าจาก element id ที่มีอยู่
  if (document.getElementById('product-list')) {
    initProductPage();
  }

  if (document.getElementById('orderForm')) {
    initOrderPage();
  }

  if (document.getElementById('ordersTable')) {
    initAdminPage();
  }
});

/* ==========================================================================
   1. หน้าสินค้า (product.html)
   ========================================================================== */
async function initProductPage() {
  const filterBar = document.getElementById('filter-bar');
  const productList = document.getElementById('product-list');
  let allProducts = [];

  // อ่านค่า ?mood=xxx จาก URL parameter
  const urlParams = new URLSearchParams(window.location.search);
  const initialMood = (urlParams.get('mood') || 'all').toLowerCase();

  try {
    const response = await fetch('products.json');
    if (!response.ok) throw new Error('ไม่สามารถโหลดไฟล์ products.json ได้');
    allProducts = await response.json();

    renderFilterButtons(filterBar, initialMood, (selectedMood) => {
      renderProducts(productList, allProducts, selectedMood);
    });

    renderProducts(productList, allProducts, initialMood);
  } catch (error) {
    console.error(error);
    productList.innerHTML = `<p style="text-align:center; color:var(--text-secondary); grid-column:1/-1;">ไม่สามารถโหลดข้อมูลสินค้าได้ในขณะนี้</p>`;
  }
}

function renderFilterButtons(container, activeMood, onFilterChange) {
  if (!container) return;

  const moods = [
    { key: 'all', label: 'ทั้งหมด' },
    { key: 'fresh', label: 'Fresh' },
    { key: 'sweet', label: 'Sweet' },
    { key: 'confident', label: 'Confident' },
    { key: 'romance', label: 'Romance' }
  ];

  container.innerHTML = moods.map(m => {
    const isActive = m.key === activeMood ? 'active' : '';
    const dot = m.key !== 'all' ? `<span class="mood-dot"></span>` : '';
    return `
      <button type="button" class="filter-btn ${isActive}" data-mood="${m.key}">
        ${dot}${m.label}
      </button>
    `;
  }).join('');

  container.querySelectorAll('.filter-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      container.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      onFilterChange(btn.dataset.mood);
    });
  });
}

function renderProducts(container, products, moodFilter) {
  const filtered = moodFilter === 'all'
    ? products
    : products.filter(p => p.mood.toLowerCase() === moodFilter);

  if (filtered.length === 0) {
    container.innerHTML = `<p style="text-align:center; color:var(--text-secondary); grid-column:1/-1;">ไม่พบสินค้าในหมวดหมู่นี้</p>`;
    return;
  }

  container.innerHTML = filtered.map(product => {
    // รวมชื่อสินค้าพร้อมขนาดเพื่อให้ระบุรายการได้ชัดเจนในหน้าสั่งซื้อ
    const fullItemName = `${product.name} ${product.size}`;
    const orderUrl = `order.html?item=${encodeURIComponent(fullItemName)}&price=${encodeURIComponent(product.price)}`;

    return `
      <article class="product-card" data-mood="${product.mood}">
        <div class="product-image-wrap">
          <img src="${product.image}" alt="${product.name}" loading="lazy">
          <span class="product-size-badge">${product.size}</span>
        </div>
        <div class="product-info">
          <div class="product-meta">
            <span class="mood-tag">
              <span class="mood-dot"></span>${product.mood}
            </span>
            <span class="product-type">${product.type === 'spray' ? 'Body Mist' : 'Roll-on'}</span>
          </div>
          <h3 class="product-title">${product.name}</h3>
          <p class="product-desc">${product.description}</p>
          <div class="product-footer">
            <span class="product-price">฿${product.price}</span>
            <a href="${orderUrl}" class="btn btn-sm">สั่งซื้อ</a>
          </div>
        </div>
      </article>
    `;
  }).join('');
}

/* ==========================================================================
   2. หน้าสั่งซื้อ (order.html)
   ========================================================================== */
function initOrderPage() {
  const orderForm = document.getElementById('orderForm');
  const itemsInput = document.getElementById('items');
  const totalInput = document.getElementById('total');

  // อ่านค่า item และ price จาก URL parameter แล้วเติมลงฟอร์มทันที
  const urlParams = new URLSearchParams(window.location.search);
  const itemParam = urlParams.get('item');
  const priceParam = urlParams.get('price');

  if (itemParam && itemsInput) {
    itemsInput.value = itemParam;
  }

  if (priceParam && totalInput) {
    totalInput.value = priceParam;
  }

  orderForm.addEventListener('submit', (e) => {
    e.preventDefault();

    const submitBtn = orderForm.querySelector('button[type="submit"]');
    if (submitBtn) {
      submitBtn.disabled = true;
      submitBtn.textContent = 'กำลังส่งข้อมูล...';
    }

    const payload = {
      customerName: document.getElementById('customerName').value.trim(),
      contact: document.getElementById('contact').value.trim(),
      items: document.getElementById('items').value.trim(),
      total: document.getElementById('total').value.trim(),
      note: document.getElementById('note').value.trim()
    };

    fetch(APPS_SCRIPT_URL, {
      method: 'POST',
      body: JSON.stringify(payload)
    })
      .then(() => {
        window.location.href = 'thankyou.html';
      })
      .catch(error => {
        console.error(error);
        alert('เกิดข้อผิดพลาด กรุณาลองใหม่อีกครั้ง');
        if (submitBtn) {
          submitBtn.disabled = false;
          submitBtn.textContent = 'ยืนยันสั่งซื้อ';
        }
      });
  });
}

/* ==========================================================================
   3. หน้าแอดมิน (admin.html)
   ========================================================================== */
async function initAdminPage() {
  const tbody = document.querySelector('#ordersTable tbody');
  if (!tbody) return;

  tbody.innerHTML = `<tr><td colspan="6" style="text-align:center; padding:2rem;">กำลังโหลดข้อมูลคำสั่งซื้อ...</td></tr>`;

  try {
    const response = await fetch(CSV_URL);
    if (!response.ok) throw new Error('ไม่สามารถดึงข้อมูล CSV ได้');

    const csvText = await response.text();
    const rows = parseCSV(csvText);

    // ตัดแถวหัวตาราง (index 0) และกรองแถวว่างออก
    const dataRows = rows.slice(1).filter(row => row.some(cell => cell.trim() !== ''));

    if (dataRows.length === 0) {
      tbody.innerHTML = `<tr><td colspan="6" style="text-align:center; padding:2rem;">ยังไม่มีรายการสั่งซื้อ</td></tr>`;
      return;
    }

    // เรียงจากรายการล่าสุดขึ้นก่อน (แถวท้ายสุดของ Google Sheet คือรายการล่าสุด)
    const latestFirst = dataRows.reverse();

    tbody.innerHTML = latestFirst.map(cols => {
      const timestamp = escapeHTML(cols[0] || '-');
      const customerName = escapeHTML(cols[1] || '-');
      const contact = escapeHTML(cols[2] || '-');
      const items = escapeHTML(cols[3] || '-');
      const total = escapeHTML(cols[4] || '-');
      const note = escapeHTML(cols[5] || '-');

      return `
        <tr>
          <td>${timestamp}</td>
          <td>${customerName}</td>
          <td>${contact}</td>
          <td>${items}</td>
          <td>฿${total}</td>
          <td>${note}</td>
        </tr>
      `;
    }).join('');
  } catch (error) {
    console.error(error);
    tbody.innerHTML = `<tr><td colspan="6" style="text-align:center; padding:2rem; color:var(--mood-romance);">เกิดข้อผิดพลาดในการโหลดข้อมูล</td></tr>`;
  }
}

// ฟังก์ชัน Parse CSV แบบเขียนเอง รองรับเครื่องหมายคำพูด "..." และขึ้นบรรทัดใหม่ภายในช่อง
function parseCSV(text) {
  const rows = [];
  let currentRow = [];
  let currentCell = '';
  let insideQuotes = false;

  for (let i = 0; i < text.length; i++) {
    const char = text[i];
    const nextChar = text[i + 1];

    if (char === '"') {
      if (insideQuotes && nextChar === '"') {
        // กรณี "" ภายในข้อความ ให้แปลงเป็น " หนึ่งตัว
        currentCell += '"';
        i++;
      } else {
        insideQuotes = !insideQuotes;
      }
    } else if (char === ',' && !insideQuotes) {
      currentRow.push(currentCell);
      currentCell = '';
    } else if ((char === '\n' || (char === '\r' && nextChar === '\n')) && !insideQuotes) {
      if (char === '\r') i++;
      currentRow.push(currentCell);
      rows.push(currentRow);
      currentRow = [];
      currentCell = '';
    } else {
      currentCell += char;
    }
  }

  // เก็บข้อมูลเซลล์/แถวสุดท้ายหากไม่ได้ลงท้ายด้วย \n
  if (currentCell !== '' || currentRow.length > 0) {
    currentRow.push(currentCell);
    rows.push(currentRow);
  }

  return rows;
}

// ป้องกัน XSS เมื่อแสดงข้อมูลที่ผู้ใช้กรอกลงในตาราง Admin
function escapeHTML(str) {
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}
